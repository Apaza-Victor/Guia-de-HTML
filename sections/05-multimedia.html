Multimedia

Imagenes

HTML
<img src="foto.jpg" alt="Descripcion" width="800" height="600"> <!-- Imagen básica: src=ruta, alt=texto alternativo, dimensiones -->

<!-- Imagen responsiva con multiples resoluciónes -->
<img src="foto.jpg"
     srcset="foto-480.jpg 480w, foto-800.jpg 800w, foto-1200.jpg 1200w" <!-- Opciones de imagen por ancho -->
     sizes="(max-width: 600px) 480px, (max-width: 1000px) 800px, 1200px" <!-- Tamano segun viewport -->
     loading="lazy"> <!-- Carga diferida (mejora rendimiento) -->

Picture para imagenes adaptables

HTML
<picture> <!-- Contenedor para imagenes adaptables por formato -->
  <source srcset="imagen.webp" type="image/webp"> <!-- Preferido: formato WebP (mas ligero) -->
  <source srcset="imagen.jpg" type="image/jpeg"> <!-- Fallback: formato JPEG -->
  <img src="imagen.jpg" alt="Descripcion"> <!-- Imagen por defecto si no soporta picture -->
</picture>

Video

HTML
<video src="video.mp4" controls width="640" height="360"> <!-- Video con controles del navegador -->
  <track kind="subtitles" src="subs.vtt" srclang="es" label="Espanol"> <!-- Subtitulos en espanol -->
  Tu navegador no soporta video HTML5. <!-- Mensaje de fallback -->
</video>

<!-- Video con multiples fuentes y autoplay -->
<video autoplay muted loop playsinline poster="poster.jpg"> <!-- autoplay=reproducir, muted=sin sonido (requerido), loop=bucle, playsinline=en linea -->
  <source src="video.webm" type="video/webm"> <!-- Formato preferido: WebM -->
  <source src="video.mp4" type="video/mp4"> <!-- Fallback: MP4 -->
</video>

Audio

HTML
<audio controls> <!-- Reproductor de audio con controles -->
  <source src="audio.mp3" type="audio/mpeg"> <!-- Formato MP3 (mayor compatibilidad) -->
  <source src="audio.ogg" type="audio/ogg"> <!-- Formato OGG (alternativa libre) -->
  Tu navegador no soporta audio HTML5. <!-- Mensaje de fallback -->
</audio>
