Drag and Drop

JavaScript
const draggable = document.querySelector('.draggable'); // Selecciona el elemento arrastrable
draggable.addEventListener('dragstart', (e) => { // Escucha el inicio del arrastre
  e.dataTransfer.setData('text/plain', e.target.id); // Guarda el ID del elemento arrastrado
});

const dropzone = document.querySelector('.dropzone'); // Selecciona la zona de destino
dropzone.addEventListener('dragover', (e) => e.preventDefault()); // Permite soltar elementos
dropzone.addEventListener('drop', (e) => { // Maneja el evento de soltar
  e.preventDefault(); // Evita el comportamiento por defecto
  const id = e.dataTransfer.getData('text/plain'); // Recupera el ID del elemento
  dropzone.appendChild(document.getElementById(id)); // Mueve el elemento a la zona destino
});
