Intersection Observer

JavaScript
const observer = new IntersectionObserver((entries) => { // Crea un observador de interseccion
  entries.forEach(entry => { // Recorre cada elemento observado
    if (entry.isIntersecting) { // Verifica si el elemento es visible
      entry.target.classList.add('visible'); // Agrega la clase visible al elemento
      observer.unobserve(entry.target); // Deja de observar ese elemento
    }
  });
}, { threshold: 0.1, rootMargin: '0px 0px -50px 0px' }); // 10% visible, margen inferior negativo

document.querySelectorAll('.animate-on-scroll').forEach(el => {
  observer.observe(el); // Observa cada elemento con la clase animate-on-scroll
});
