Meta Tags y SEO

HTML
<head> <!-- Inicio del bloque de metadatos -->
  <meta charset="UTF-8"> <!-- Define la codificacion de caracteres -->
  <meta name="viewport" content="width=device-width, initial-scale=1.0"> <!-- Diseno responsivo -->
  <title>Mi Pagina - Titulo Descriptivo</title> <!-- Titulo para navegadores y buscadores -->
  <meta name="description" content="Descripcion de 150-160 caracteres"> <!-- Descripcion para SEO -->
  <meta name="robots" content="index, follow"> <!-- Indica a los buscadores indexar y rastrear -->
  <link rel="canonical" href="https://ejemplo.com/pagina"> <!-- URL canonica para evitar contenido duplicado -->

  <!-- Open Graph (redes sociales) -->
  <meta property="og:title" content="Titulo"> <!-- Titulo para compartir en redes -->
  <meta property="og:description" content="Descripcion"> <!-- Descripcion para compartir -->
  <meta property="og:image" content="https://ejemplo.com/imagen.jpg"> <!-- Imagen de vista previa -->
  <meta property="og:url" content="https://ejemplo.com"> <!-- URL canonica de la página -->

  <!-- Twitter Card -->
  <meta name="twitter:card" content="summary_large_image"> <!-- Tipo de tarjeta en Twitter -->
  <meta name="twitter:title" content="Titulo"> <!-- Titulo para Twitter -->

  <link rel="preconnect" href="https://fonts.googleapis.com"> <!-- Preconectar para cargar fuentes mas rapido -->
  <meta name="theme-color" content="#6366f1"> <!-- Color del tema en moviles -->
</head> <!-- Fin del bloque head -->
