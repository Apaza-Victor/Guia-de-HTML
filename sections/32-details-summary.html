Details, Summary y Dialog

HTML
<details> <!-- Contenido desplegable/colapsable -->
  <summary>Que es HTML5?</summary> <!-- Texto visible que activa el toggle -->
  <p>HTML5 es la quinta revision del lenguaje HTML.</p> <!-- Contenido oculto hasta hacer clic -->
</details>

<dialog id="modal"> <!-- Elemento dialog nativo para modales -->
  <form method="dialog"> <!-- Formulario que cierra el dialog al enviar -->
    <h2>Confirmar</h2>
    <p>Estas seguro?</p>
    <button value="cancel">No</button> <!-- Boton que cierra con valor cancel -->
    <button value="confirm">Si</button> <!-- Boton que confirma con valor confirm -->
  </form>
</dialog>
