Canvas API

HTML + JavaScript
<canvas id="miCanvas" width="800" height="600"></canvas> <!-- Elemento canvas para dibujar -->

<script>
const canvas = document.getElementById('miCanvas'); // Obtiene el elemento canvas del DOM
const ctx = canvas.getContext('2d'); // Obtiene el contexto 2D para dibujar

// Rectangulo
ctx.fillStyle = '#6366f1'; // Define el color de relleno
ctx.fillRect(50, 50, 200, 100); // Dibuja un rectangulo (x, y, ancho, alto)

// Circulo
ctx.beginPath(); // Inicia un nuevo trazo
ctx.arc(400, 100, 50, 0, Math.PI * 2); // Dibuja un arco circular completo
ctx.fillStyle = '#10b981'; // Cambia el color a verde
ctx.fill(); // Rellena la forma dibujada

// Texto
ctx.font = 'bold 24px Inter'; // Define la fuente del texto
ctx.fillStyle = '#0f172a'; // Color del texto
ctx.fillText('Hola Canvas!', 50, 400); // Escribe texto en el canvas
</script>

Animacion con requestAnimationFrame

JavaScript
let x = 0; // Posicion inicial del circulo en el eje X
function animar() {
  ctx.clearRect(0, 0, canvas.width, canvas.height); // Limpia todo el canvas
  ctx.beginPath(); // Inicia un nuevo trazo para el circulo
  ctx.arc(x, 150, 30, 0, Math.PI * 2); // Dibuja circulo en la posición actual
  ctx.fillStyle = '#6366f1'; // Color morado para el circulo
  ctx.fill(); // Rellena el circulo
  x += 2; // Mueve el circulo 2 pixeles a la derecha
  if (x > canvas.width) x = 0; // Reinicia si sale del canvas
  requestAnimationFrame(animar); // Programa el siguiente frame de animacion
}
animar(); // Inicia la animacion
