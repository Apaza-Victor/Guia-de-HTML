Fullscreen API

Pantalla completa para cualquier elemento

La Fullscreen API permite expandir cualquier elemento HTML a pantalla completa, util para videos, juegos, presentaciones y visualizaciones inmersivas.

JavaScript
// Activar pantalla completa en un elemento
function pantallaCompleta(elemento) {
  if (elemento.requestFullscreen) {
    elemento.requestFullscreen();
  } else if (elemento.webkitRequestFullscreen) {
    elemento.webkitRequestFullscreen(); // Safari
  }
}

// Salir de pantalla completa
function salirPantallaCompleta() {
  if (document.exitFullscreen) {
    document.exitFullscreen();
  }
}

// Detectar cambios de estado
document.addEventListener('fullscreenchange', () => {
  if (document.fullscreenElement) {
    console.log('En pantalla completa');
  } else {
    console.log('Salio de pantalla completa');
  }
});

// Verificar si esta en pantalla completa
const enFullscreen = !!document.fullscreenElement;
