API de Geolocalización

JavaScript
if ('geolocation' in navigator) { // Verifica si el navegador soporta geolocalización
  navigator.geolocation.getCurrentPosition( // Solicita la posición actual del usuario
    (position) => { // Callback de exito con la posición
      const { latitude, longitude } = position.coords; // Extrae latitud y longitud
      console.log(`Lat: ${latitude}, Lng: ${longitude}`); // Muestra las coordenadas
    },
    (error) => { // Callback de error si falla la geolocalización
      console.error('Error:', error.message); // Muestra el mensaje de error
    },
    { enableHighAccuracy: true, timeout: 10000 } // Opciones: alta precision, timeout de 10s
  );
}
