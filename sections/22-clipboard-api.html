Clipboard API

Copiar y pegar texto de forma moderna y segura

La Clipboard API reemplaza al obsoleto document.execCommand('copy') con una interfaz basada en Promesas que funciona en安全 modo seguro bajo HTTPS.

JavaScript
// Copiar texto al portapapeles
async function copiarTexto(texto) {
  try {
    await navigator.clipboard.writeText(texto);
    console.log('Texto copiado exitosamente');
  } catch (err) {
    console.error('Error al copiar:', err);
  }
}

// Leer texto del portapapeles
async function pegarTexto() {
  try {
    const texto = await navigator.clipboard.readText();
    console.log('Texto pegado:', texto);
  } catch (err) {
    console.error('Error al pegar:', err);
  }
}

// Copiar con feedback visual
async function copiarConFeedback(btn, texto) {
  await navigator.clipboard.writeText(texto);
  const original = btn.textContent;
  btn.textContent = '¡Copiado!';
  btn.style.background = '#22c55e';
  setTimeout(() => {
    btn.textContent = original;
    btn.style.background = '';
  }, 2000);
}
Requisitos de seguridad

La Clipboard API solo funciona en contextos seguros (HTTPS o localhost) y requiere que el usuario interactue con la pagina (click, keypress) para autorizar el acceso.
