Template y Web Components

JavaScript
class MiBoton extends HTMLElement { // Define un Web Component heredando de HTMLElement
  constructor() {
    super(); // Llama al constructor del elemento padre
    const shadow = this.attachShadow({ mode: 'open' }); // Crea un Shadow DOM encapsulado
    shadow.innerHTML = `
      <style>
        button { padding: 8px 16px; background: #6366f1;
                 color: white; border: none; border-radius: 6px; cursor: pointer; }
        button:hover { background: #4f46e5; }
      </style>
      <button><slot>Click</slot></button>
    `; // Estilos y HTML encapsulados dentro del Shadow DOM
  }
}
customElements.define('mi-boton', MiBoton); // Registra el custom element en el navegador

Uso: <mi-boton>Mi Texto</mi-boton>
