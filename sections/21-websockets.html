WebSockets

JavaScript
const socket = new WebSocket('wss://ejemplo.com/chat'); // Abre conexión WebSocket segura
socket.addEventListener('open', () => { // Escucha cuando se establece la conexión
  socket.send(JSON.stringify({ type: 'message', content: 'Hola!' })); // Envia un mensaje al servidor
});
socket.addEventListener('message', (event) => { // Escucha mensajes entrantes del servidor
  const data = JSON.parse(event.data); // Convierte el JSON recibido a objeto
  console.log('Recibido:', data); // Muestra los datos recibidos
});
