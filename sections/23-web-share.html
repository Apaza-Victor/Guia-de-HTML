Web Share API

Comparte contenido usando el dialogo nativo del sistema

La Web Share API abre el dialogo de compartir nativo del sistema operativo (Android, iOS, Windows) permitiendo compartir texto, URLs y archivos con cualquier app instalada.

JavaScript
// Compartir texto simple
async function compartir() {
  if ('share' in navigator) {
    try {
      await navigator.share({
        title: 'Guia HTML5',
        text: 'Mira esta guia completa de HTML5',
        url: 'https://ejemplo.com/html5'
      });
    } catch (err) {
      if (err.name !== 'AbortError') {
        console.error('Error al compartir:', err);
      }
    }
  } else {
    // Fallback: copiar URL al portapapeles
    await navigator.clipboard.writeText('https://ejemplo.com/html5');
  }
}

// Compartir archivos
async function compartirArchivo(archivo) {
  if ('share' in navigator && 'canShare' in navigator) {
    const datos = {
      title: 'Mi archivo',
      files: [archivo]
    };
    if (navigator.canShare(datos)) {
      await navigator.share(datos);
    }
  }
}
