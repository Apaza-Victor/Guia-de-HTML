Funciones

JavaScript
// Funcion tradicional
function sumar(a, b) { return a + b; } // Declara una función con parametros y retorno

// Arrow function
const sumar = (a, b) => a + b; // Sintaxis abreviada de función en ES6

// Async/await
async function obtenerDatos() {
  const res = await fetch('/api/datos'); // Espera la respuesta de la API

  return res.json(); // Devuelve los datos en formato JSON
}

Arrays

JavaScript
const nums = [1, 2, 3, 4, 5]; // Array de numeros para operaciónes

nums.map(n => n * 2);           // Transforma cada elemento: duplica los valores
nums.filter(n => n > 3);      // Filtra: devuelve solo mayores a 3
nums.reduce((a, b) => a + b, 0); // Reduce: suma todos los elementos
nums.find(n => n > 2);        // Busca: devuelve el primer elemento mayor a 2
nums.some(n => n > 4);        // Verifica: true si algun elemento es mayor a 4

DOM Manipulation

JavaScript
document.querySelector('.clase'); // Selecciona el primer elemento que coincida con el selector
document.querySelectorAll('p'); // Selecciona todos los elementos que coincidan
elemento.addEventListener('click', () => {}); // Registra un evento click en el elemento
elemento.classList.add('active'); // Agrega una clase CSS al elemento
elemento.style.color = 'red'; // Cambia el estilo CSS directamente
elemento.innerHTML = '<strong>Nuevo</strong>'; // Reemplaza el contenido HTML interno
