Accesibilidad y ARIA

HTML
<!-- aria-label -->
<button aria-label="Cerrar menu">&times;</button> <!-- Label accesible para lectores de pantalla -->

<!-- aria-labelledby -->
<div aria-labelledby="titulo"> <!-- Vincula el div a un titulo existente -->
  <h2 id="titulo">Seccion</h2>
</div> <!-- Fin del div con aria-labelledby -->

<!-- aria-live -->
<div aria-live="polite">Contenido dinamico</div> <!-- Anuncia cambios a lectores de pantalla -->

<!-- tabindex -->
<input tabindex="0"> <!-- Permite navegación por teclado con Tab -->
