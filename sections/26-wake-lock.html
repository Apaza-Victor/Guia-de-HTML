Screen Wake Lock API

Evita que la pantalla se apague

La Screen Wake Lock API mantiene la pantalla encendida mientras el usuario esta interactuando con tu app, ideal para recetas, lectura, videollamadas o presentaciones.

JavaScript
let wakeLock = null;

// Activar wake lock
async function activarWakeLock() {
  if ('wakeLock' in navigator) {
    try {
      wakeLock = await navigator.wakeLock.request('screen');
      console.log('Wake Lock activado');

      // Se libera automaticamente al cambiar de pestana
      wakeLock.addEventListener('release', () => {
        console.log('Wake Lock liberado');
      });
    } catch (err) {
      console.error('Error:', err.message);
    }
  }
}

// Liberar manualmente
function liberarWakeLock() {
  if (wakeLock) {
    wakeLock.release();
    wakeLock = null;
  }
}

// Reactivar al volver a la pestana
document.addEventListener('visibilitychange', async () => {
  if (document.visibilityState === 'visible') {
    await activarWakeLock();
  }
});
